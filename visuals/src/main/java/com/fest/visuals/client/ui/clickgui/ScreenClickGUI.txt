package com.fest.visuals.client.ui.clickgui;

import com.mojang.blaze3d.vertex.PoseStack;
import net.minecraft.client.gui.GuiGraphicsExtractor;
import net.minecraft.client.gui.screens.Screen;
import net.minecraft.network.chat.Component;
import org.lwjgl.glfw.GLFW;

import com.fest.visuals.api.utils.animation.Easing;
import com.fest.visuals.api.utils.animation.AnimationUtil;
import com.fest.visuals.api.utils.render.RenderUtil;
import com.fest.visuals.api.utils.render.ScissorUtil;
import com.fest.visuals.client.ui.clickgui.ClickGuiConfigs;
import com.fest.visuals.client.ui.clickgui.ClickGuiModuleList;
import com.fest.visuals.client.ui.theme.ThemeEditor;
import com.fest.visuals.client.ui.clickgui.ClickGuiWaypoints;
import com.fest.visuals.api.system.interfaces.QuickImports;

import java.awt.Color;

public class ScreenClickGUI extends Screen implements QuickImports {
    private static ScreenClickGUI instance;

    public static ScreenClickGUI getInstance() {
        if (instance == null) instance = new ScreenClickGUI();
        return instance;
    }

    private final AnimationUtil openAnimation = new AnimationUtil();
    private final AnimationUtil contentAnimation = new AnimationUtil();

    private final ClickGuiHeader header = new ClickGuiHeader();
    private final ClickGuiSidebar sidebar = new ClickGuiSidebar();
    private final ClickGuiModuleList modules = new ClickGuiModuleList();
    private final ClickGuiConfigs configs = new ClickGuiConfigs();
    private final ClickGuiWaypoints waypointsUI = new ClickGuiWaypoints();
    private final ThemeEditor themeEditor = new ThemeEditor();

    private ClickGuiTab tab = ClickGuiTab.RENDER;
    private ClickGuiTab pendingTab = ClickGuiTab.RENDER;
    private boolean switching = false;
    private boolean closing = false;

    public ScreenClickGUI() {
        super(Component.literal("FestVisuals GUI"));
        openAnimation.setValue(0.0);
        contentAnimation.setValue(1.0);
    }

    @Override
    protected void init() {
        openAnimation.setValue(0.0);
        closing = false;
        themeEditor.setOpen(tab == ClickGuiTab.THEME);
        themeEditor.setEmbedded(tab == ClickGuiTab.THEME);
    }

    @Override
    public void onClose() {
        configs.close();
        waypointsUI.close();
        themeEditor.setOpen(false);
    }

    @Override
    public void extractRenderState(GuiGraphicsExtractor context, int mouseX, int mouseY, float delta) {
        super.extractRenderState(context, mouseX, mouseY, delta);

        if (themeEditor.isOpen() && tab != ClickGuiTab.THEME && !switching) {
            requestSwitch(ClickGuiTab.THEME);
        }

        openAnimation.update();
        openAnimation.run(closing ? 0.0 : 1.0, 300, Easing.EXPO_OUT);
        float openAnim = (float) openAnimation.getValue();

        if (closing && openAnim <= 0.01f) {
            mc.gui.setScreen(null);
            return;
        }

        float windowX = mc.getWindow().getGuiScaledWidth() / 2f - ClickGuiLayout.windowWidth() / 2f;
        float windowY = mc.getWindow().getGuiScaledHeight() / 2f - ClickGuiLayout.windowHeight() / 2f;

        PoseStack matrices = RenderUtil.matrices();
        RenderUtil.RECT.draw(matrices, 0, 0, mc.getWindow().getGuiScaledWidth(), mc.getWindow().getGuiScaledHeight(), 0f, new Color(0, 0, 0, (int) (140 * openAnim)));
        RenderUtil.RECT.draw(matrices, windowX, windowY, ClickGuiLayout.windowWidth(), ClickGuiLayout.windowHeight(), ClickGuiLayout.round(), new Color(16, 16, 18, (int) (openAnim * 255f)));

        float content = (float) contentAnimation.getValue();
        header.render(context, windowX, windowY, openAnim, content, tab, tab == ClickGuiTab.THEME);
        sidebar.render(context, windowX, windowY, openAnim, switching ? pendingTab : tab);

        ScissorUtil.start(matrices, ClickGuiLayout.contentX(windowX), ClickGuiLayout.contentY(windowY), ClickGuiLayout.contentWidth(), ClickGuiLayout.contentHeight());
        
        if (!switching) {
            contentAnimation.update();
            contentAnimation.run(1.0, 180, Easing.EXPO_OUT);
            
            if (tab == ClickGuiTab.THEME) {
                themeEditor.setWidth(ClickGuiLayout.contentWidth());
                themeEditor.setHeight(ClickGuiLayout.contentHeight() - ClickGuiLayout.scaled(16f));
                themeEditor.render(context, mouseX, mouseY, delta);
            } else if (tab == ClickGuiTab.CONFIGS && sidebar.query().isEmpty()) {
                configs.render(context, windowX, windowY, content, mouseX, mouseY);
            } else if (tab == ClickGuiTab.WAYPOINTS && sidebar.query().isEmpty()) {
                waypointsUI.render(context, windowX, windowY, content, mouseX, mouseY);
            } else {
                modules.render(context, windowX, windowY, content, tab, sidebar.query(), mouseX, mouseY, delta);
            }
        } else {
            contentAnimation.update();
            contentAnimation.run(0.0, 120, Easing.SINE_OUT);
            if (contentAnimation.getValue() <= 0.05) {
                tab = pendingTab;
                themeEditor.setOpen(tab == ClickGuiTab.THEME);
                themeEditor.setEmbedded(tab == ClickGuiTab.THEME);
                if (tab == ClickGuiTab.CONFIGS) configs.refresh();
                else configs.close();
                if (tab != ClickGuiTab.WAYPOINTS) waypointsUI.close();
                switching = false;
            }
        }

        ScissorUtil.stop(matrices);
    }

    private void requestSwitch(ClickGuiTab next) {
        if (!switching && next == tab && next != ClickGuiTab.THEME) {
            modules.resetScroll();
            return;
        }
        pendingTab = next;
        switching = true;
        sidebar.resetSearchFocus();
        if (next != ClickGuiTab.THEME) themeEditor.setOpen(false);
    }

    @Override
    public boolean keyPressed(int keyCode, int scanCode, int modifiers) {
        if (sidebar.keyPressed(keyCode)) {
            return true;
        }

        if (tab == ClickGuiTab.THEME) {
            themeEditor.keyPressed(keyCode, scanCode, modifiers);
            return true;
        }

        if (keyCode == GLFW.GLFW_KEY_ESCAPE) {
            closing = true;
            return true;
        }
        return super.keyPressed(keyCode, scanCode, modifiers);
    }

    @Override
    public boolean charTyped(char chr, int modifiers) {
        if (sidebar.charTyped(chr)) return true;
        if (tab == ClickGuiTab.THEME) {
            return themeEditor.charTyped(chr, modifiers);
        }
        return super.charTyped(chr, modifiers);
    }

    @Override
    public boolean mouseClicked(double mouseX, double mouseY, int button) {
        float windowX = mc.getWindow().getGuiScaledWidth() / 2f - ClickGuiLayout.windowWidth() / 2f;
        float windowY = mc.getWindow().getGuiScaledHeight() / 2f - ClickGuiLayout.windowHeight() / 2f;

        if (button == 0 && header.themeClicked(mouseX, mouseY, windowX, windowY)) {
            requestSwitch(ClickGuiTab.THEME);
            return true;
        }

        ClickGuiTab sidebarHit = sidebar.clickedTab(mouseX, mouseY, windowX, windowY);
        if (sidebarHit != null && sidebarHit != tab) {
            requestSwitch(sidebarHit);
            return true;
        }

        if (tab == ClickGuiTab.THEME) {
            themeEditor.mouseClicked(mouseX, mouseY, button);
            return true;
        }

        if (tab == ClickGuiTab.CONFIGS && sidebar.query().isEmpty()) {
            return configs.mouseClicked(mouseX, mouseY, windowX, windowY) || super.mouseClicked(mouseX, mouseY, button);
        }

        if (tab == ClickGuiTab.WAYPOINTS && sidebar.query().isEmpty()) {
            return waypointsUI.mouseClicked(mouseX, mouseY, windowX, windowY) || super.mouseClicked(mouseX, mouseY, button);
        }

        modules.mouseClicked(mouseX, mouseY, button, windowX, windowY, tab, sidebar.query());

        return super.mouseClicked(mouseX, mouseY, button);
    }

    @Override
    public boolean mouseReleased(double mouseX, double mouseY, int button) {
        float windowX = mc.getWindow().getGuiScaledWidth() / 2f - ClickGuiLayout.windowWidth() / 2f;
        float windowY = mc.getWindow().getGuiScaledHeight() / 2f - ClickGuiLayout.windowHeight() / 2f;
        if (tab == ClickGuiTab.THEME) themeEditor.mouseReleased(mouseX, mouseY, button);
        else modules.mouseReleased(mouseX, mouseY, button, tab, sidebar.query());
        return super.mouseReleased(mouseX, mouseY, button);
    }

    @Override
    public boolean mouseScrolled(double mouseX, double mouseY, double horizontalAmount, double verticalAmount) {
        float windowX = mc.getWindow().getGuiScaledWidth() / 2f - ClickGuiLayout.windowWidth() / 2f;
        float windowY = mc.getWindow().getGuiScaledHeight() / 2f - ClickGuiLayout.windowHeight() / 2f;
        
        if (tab == ClickGuiTab.THEME) {
            themeEditor.mouseScrolled(mouseX, mouseY, horizontalAmount, verticalAmount);
            return true;
        }
        
        if (tab == ClickGuiTab.CONFIGS && sidebar.query().isEmpty()) {
            return configs.mouseScrolled(mouseX, mouseY, verticalAmount, windowX, windowY);
        }
        if (tab == ClickGuiTab.WAYPOINTS && sidebar.query().isEmpty()) {
            return waypointsUI.mouseScrolled(mouseX, mouseY, verticalAmount, windowX, windowY);
        }
        return modules.mouseScrolled(mouseX, mouseY, verticalAmount, windowX, windowY, tab, sidebar.query());
    }
}
